







TEST(Assembler, jumpSignTestTrue){
    std::array<std::array<bool, 16>, 256> assemblerMemory = assembler("AssemblerCodeTest.txt");

    CPU cpuTest;
    cpuTest.instructionMemory = assemblerMemory;

    cpuTest.setRegister(std::array<bool, 8> {1, 1, 1, 1, 1, 1, 0, 1}, 0);
    cpuTest.setRegister(std::array<bool, 8> {1, 1, 1, 1, 1, 1, 0, 1}, 1);

    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();
    cpuTest.iterateCPU();

    std::array<bool, 8> output = cpuTest.getAccumulator();
    std::array<bool, 8> expected = {1, 1, 1, 1, 1, 0, 1, 1};
    EXPECT_EQ(output, expected);
}
